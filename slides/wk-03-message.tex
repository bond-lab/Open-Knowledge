\section{Why should we write?}

\begin{frame}{Why Write?}
\begin{itemize}
  \item Writing is a tool for thinking as well as a means of communication
  \item It clarifies ideas, interprets texts, and joins a scholarly conversation
  \item Written work demonstrates:
    \begin{itemize}
      \item Understanding of key issues
      \item Ability to argue persuasively
      \item Awareness of disciplinary methods
    \end{itemize}
  \item Writing is a way of finding out what you think
\end{itemize}
\vfill
Partly based on \href{https://mlpp.pressbooks.pub/writinghandbook/chapter/chapter-1/}{Analyzing Texts, Taking Notes} \citep[Ch.~1]{writinghandbook}
\end{frame}



%------------------------------------------------------------
\begin{frame}{Humanities, Social Sciences, and Linguistics}
\begin{itemize}
  \item All three study human experience, culture, society and language
  \item \textbf{Humanities:}
    \begin{itemize}
      \item Interpret texts, artworks, languages, and histories
      \item Value close reading, interpretation, argumentation
    \end{itemize}
  \item \textbf{Social Sciences:}
    \begin{itemize}
      \item Study human behavior, institutions, and systems
      \item Employ observation, evidence, and models
    \end{itemize}
  \item \textbf{Linguistics:} a bridge between the two, both theoretical and empirical
    \begin{itemize}
      \item General: structure and use of language
      \item Variation \& change: sociolinguistics, dialectology, historical linguistics
      \item Mind \& processing: psycholinguistics, neurolinguistics
      \item Data \& methods: corpora, fieldwork/elicitation, experiments, formal modeling
      \item Technology \& applications: computational linguistics/NLP, lexicography, language documentation
    \end{itemize}
  \item All rely on critical analysis, evidence, and writing suited to the audience and genre
\end{itemize}
\end{frame}


%------------------------------------------------------------
\begin{frame}{Writing as Inquiry}
\begin{itemize}
  \item Writing generates ideas and sharpens questions
  \item Early drafts explore; conclusions come later
  \item Revision is discovery: each draft deepens understanding
  \item Balance open exploration with a focused argument
  \item Thinking happens through writing, not before it
\end{itemize}
\end{frame}

%------------------------------------------------------------
\begin{frame}{Academic Conversations}
\begin{itemize}
  \item Academic writing joins a conversation that started before you
  \item You engage with others by:
    \begin{itemize}
      \item Quoting and analyzing sources
      \item Summarizing and synthesizing prior work
      \item Acknowledging different viewpoints
    \end{itemize}
  \item An essay responds to that conversation and adds to it
  \item Citations give credit for others' work, and let readers check it
\end{itemize}
\end{frame}

%------------------------------------------------------------

%------------------------------------------------------------

%------------------------------------------------------------

%------------------------------------------------------------

%------------------------------------------------------------
\begin{frame}{Voice and Style}
\begin{itemize}
  \item Academic writing has a clear, confident voice
  \item Strive for:
    \begin{itemize}
      \item Precision over ornamentation
      \item Clarity over complexity
      \item Variety in sentence structure
    \end{itemize}
  \item Avoid jargon unless necessary
  \item Use active verbs and concise phrasing
  \item Essays need a logical progression of ideas
    \begin{itemize}
    \item Use an outline to keep on track
    \item Check that every section supports the thesis
    \end{itemize}
\end{itemize}
\end{frame}

% ------------------------------------------------------------
\begin{frame}{Revising and Editing}
\begin{itemize}
  \item Revision works on the ideas; editing works on the words
  \item Strategies:
    \begin{itemize}
      \item Read aloud to test clarity
      \item Seek peer or instructor feedback
      \item Review argument flow
      \item Cut redundancy
    \end{itemize}
  \item Proofread before you submit
\end{itemize}
\end{frame}

%------------------------------------------------------------

%------------------------------------------------------------
\begin{frame}{Academic Integrity}
\begin{itemize}
  \item Cite every idea you borrow
  \item Keep detailed notes on sources
  \item A paraphrase restates the idea in your own words; swapping a few words is still copying
  \item Readers trust work when they can see where it came from
\end{itemize}
\end{frame}

\subsection{Reading for Writing}
\begin{frame}{Reading for Writing}
\begin{itemize}
  \item Writing starts with reading and taking notes
  \item Treat every lecture, discussion and reading as a text to analyse
  \item Keep asking questions as you read
  \item Take notes you can turn into arguments later
  \item Work out how the text does what it does, as well as what it says
  \end{itemize}

\vfill
Partly based on \href{https://mlpp.pressbooks.pub/writinghandbook/chapter/chapter-1/}{Analyzing Texts, Taking Notes} \citep[Ch.~1]{writinghandbook}

\end{frame}

\begin{frame}{What Counts as a Text?}
\begin{itemize}
  \item Anything you meet in class: readings, lectures, discussions
  \item Analyse them all the time, not only when told to
  \item Compare new material with prior readings, lectures, and beliefs
  \item Write your thoughts down: notes are the start of the essay
\end{itemize}
\end{frame}

\begin{frame}{Active Reading Mindset}
\begin{itemize}
  \item Break the text into parts: what is each one for?
  \item Notice plot, imagery, symbolism and allusion (not only in literature)
  \item Nonfiction uses these tools too
  \item Read for patterns as well as facts
  \item Taking apart other people's writing teaches you to build your own
\end{itemize}
\end{frame}

\begin{frame}{Questions to Drive Analysis}
\begin{itemize}
  \item What confuses you? What needs clarification?
  \item Which claims are most central to the text's project?
  \item How do structure and language support those claims?
  \item What assumptions or premises are in play?
  \item How does this text relate to other course materials and your concerns?
\end{itemize}
\end{frame}

\begin{frame}{For Fiction (Mostly)}
\begin{itemize}
  \item \textbf{Narrator}: who tells the story? reliable/unreliable/biased?
  \item \textbf{Setting \& tone}: what senses and emotions are evoked?
  \item \textbf{Characters}: motivations, loyalties, whom you side with
  \item \textbf{Language/diction}: level and implications
  \item \textbf{Plot/structure}: problems, challenges, archetypes
  \item \textbf{Images/motifs}: repetitions, metaphors, patterns
  \item \textbf{Ending}: what resolves? why end there?
\end{itemize}
\end{frame}

\begin{frame}{For Nonfiction (Mostly)}
\begin{itemize}
  \item \textbf{Author}: background and qualifications
  \item \textbf{Audience}: allies, opponents, or neutral readers?
  \item \textbf{Intention}: explanatory, polemical, celebratory? Why was it written?
  \item \textbf{Structure}: how is the argument organized?
  \item \textbf{Appeals}: logic vs. emotion; what types of arguments are used?
\end{itemize}
\end{frame}

\begin{frame}{On Arguments: Classical Roots}
\begin{itemize}
  \item Aristotle's analysis of argument still works
  \item Logic is central to many, but not all, arguments
  \item A \txx{syllogism} shows how accepted premises force a conclusion
  \item Premises can be disputed
  \item Most real reasoning gives only probable conclusions
\end{itemize}
\end{frame}

\begin{frame}{Deduction}
\begin{itemize}
  \item From accepted premises to a conclusion that must follow
    \begin{itemize}
    \item All data centres use water; this is a data centre; so it uses water
    \end{itemize}
  \item If the premises hold, the conclusion holds
  \item Works when everyone accepts the same facts and definitions
  \item Limit: most debates are about the premises
  \item So state your premises, and test them
\end{itemize}
\end{frame}

\begin{frame}{Induction}
\begin{itemize}
  \item From observations to a generalisation
    \begin{itemize}
    \item Three data centres in Arizona use a lot of water; so data centres in dry places use a lot of water (how many did you check?)
    \end{itemize}
  \item Conclusions are tentative (we never observe everything)
  \item Even strong theories can be revised
  \item Good induction follows the data
  \item Match the strength of the claim to the evidence
\end{itemize}
\end{frame}

\begin{frame}{Narrative as Argument}
\begin{itemize}
  \item Stories persuade because we identify with the people in them
  \item History blends data, concepts, and narrative structure
  \item Narrative sometimes stands in for data
  \item When a writer tells a story, ask what work it is doing
\end{itemize}
\end{frame}

\begin{frame}{Reason, Emotion, and Premises}
\begin{itemize}
  \item Separate appeals to reason from appeals to emotion
  \item Trace the path from premises to conclusion
  \item Test premises for assumptions, generality, and evidence
  \item Where is the uncertainty: in the data, the inference, or the values?
\end{itemize}
\end{frame}

\begin{frame}{Note-Taking Checklist}
\begin{itemize}
  \item Capture the central question and main claims
  \item Mark evidence and how it supports claims
  \item Flag key terms, metaphors, and recurring motifs
  \item Record questions and possible counterarguments
  \item Boil it down to one or two sentences you can use later
\end{itemize}
\end{frame}

\section{What is your message?}


\begin{frame}{What Is a Thesis?}
\begin{itemize}
  \item A thesis is a claim that can be defended with reasons and evidence
  \item It is neither a topic nor a fact, but an interpretation
  \item Example:
    \begin{itemize}
      \item Topic: “Women in Shakespeare”
      \item Thesis: “Shakespeare’s comedies use disguise to challenge gender norms”
    \end{itemize}
  \item It is a promise to the reader about where the essay goes
\end{itemize}
\vfill
Partly based on \href{https://mlpp.pressbooks.pub/writinghandbook/chapter/44/}{Creating a Thesis} \citep[Ch.~3]{writinghandbook}
\end{frame}

\begin{frame}{From Question to Argument}
\begin{itemize}
  \item Start with a question you want answered
  \item Narrow it until it can be answered in the space you have
  \item Ask: how? why? so what?
  \item Your first answer is the working thesis
  \item Revise the thesis as new evidence appears
\end{itemize}
\end{frame}

\begin{frame}{Characteristics of a Strong Thesis}
\begin{itemize}
  \item \textbf{Debatable}: reasonable people could disagree
  \item \textbf{Specific}: not a vague generality
  \item \textbf{Focused}: answerable in the space you have
  \item \textbf{Insightful}: says something not already obvious
  \item \textbf{Connected}: matches the evidence you have
\end{itemize}
\vspace{1ex}
\small
\begin{tabular}{@{}p{0.46\textwidth}p{0.46\textwidth}@{}}
\toprule
\textbf{Weak} --- announces a topic & \textbf{Strong} --- takes a stance \\
\midrule
``This essay will discuss social media and teenagers.'' &
``Social media intensifies teenage anxiety by rewarding performative
identity.'' \\
\addlinespace
``This essay will discuss data centres and water.'' &
``Data centre water use is small nationally but heavy locally, so it should
be reported by watershed.'' \\
\bottomrule
\end{tabular}
\vspace{1ex}

A strong thesis makes the reader ask \textbf{how?} and \textbf{why?}
\end{frame}


\begin{frame}{Refining Your Thesis}
\begin{itemize}
  \item An early thesis is a rough hypothesis
  \item Strengthen by:
    \begin{itemize}
      \item Clarifying key terms
      \item Tightening scope
      \item Checking consistency with evidence
    \end{itemize}
  \item Ask a classmate to summarise your claim: is that what you meant?
    \begin{itemize}
    \item \textbf{That is what we are about to do with your water paragraphs}
    \end{itemize}
\end{itemize}
\end{frame}


\begin{frame}{Common Pitfalls}
\begin{itemize}
  \item Thesis too broad or too narrow
  \item Summarises instead of analysing
  \item Several unconnected claims
  \item Uses vague verbs: \textit{shows}, \textit{is about}, \textit{explores}
  \item Ignores the counterarguments
\end{itemize}
\end{frame}

\begin{frame}{Purpose and Tone Across Disciplines}
\begin{tabular}{p{3cm}p{7cm}}
\textbf{Humanities} & Persuasive and interpretive: argument from ideas and textual evidence \\[0.5em]
\textbf{Social Sciences} & Empirical and neutral: test hypotheses, describe data \\[0.5em]
\textbf{Linguistics} & Analytical: theory framed and tested against data \\[0.5em]
\textbf{Computer Science} & Technical and concise: algorithms, models, and how well they score
\end{tabular}
\end{frame}

\begin{frame}{How Arguments Are Built}
\begin{itemize}
  \item \textbf{Humanities}: evidence supports an interpretation
  \item \textbf{Social Sciences}: evidence tests a hypothesis
  \item \textbf{Linguistics}: evidence shows a pattern or a contrast
  \item \textbf{Computer Science}: results must be reproducible
  \item \textbf{All}: say clearly what you add
\end{itemize}
\end{frame}

\begin{frame}{Encyclopedic Writing: Wikipedia Style}
\begin{itemize}
  \item \textbf{Purpose}: inform, not argue; summarise what is accepted
  \item \textbf{Tone}: neutral, verifiable, non-original
  \item \textbf{Structure}: by topic (lead, body, appendices), see
    \href{https://en.wikipedia.org/wiki/Wikipedia:Manual_of_Style/Layout}{Wikipedia:Manual of Style/Layout}
  \item \textbf{Comparison}:
    \begin{itemize}
      \item No new data or interpretation (unlike a research paper)
      \item Like the introduction of a paper: context and the key sources
    \end{itemize}
\end{itemize}
\end{frame}

\section{How can you convince people?}

\begin{frame}{From Topic to Argument}
\begin{itemize}
  \item From collecting ideas to building a case
  \item Argue from reasons; use emotion sparingly
  \item Choose the kind of reasoning that fits your material
  \item State your conclusions with confidence, and their limits
  \item Let the structure show the reader your reasoning
  \end{itemize}
  \vfill
Partly based on \href{https://mlpp.pressbooks.pub/writinghandbook/chapter/ordering-evidence-building-an-argument/}{Ordering Evidence, Building an Argument} \citep[Ch.~4]{writinghandbook}
\end{frame}



\begin{frame}{Worry about the structure more than the style}
  \begin{quote}
    Prose is architecture, not interior decoration, and the Baroque is over.
    \\
    \hfill \textnormal{Ernest Hemingway (1932) \textit{Death in the afternoon}}
  \end{quote}
\begin{itemize}
  \item Foundation: a thesis and the reasons that hold it up
  \item Form follows function: the structure is there to make the argument clear
  \item The scaffolding can be invisible, but it has to be there
  \item A pile of points is not an argument
  \end{itemize}

\end{frame}


\begin{frame}{Outline: Before or After Drafting}
  \begin{itemize}
  \item Two approaches
    \begin{itemize}
    \item[A] sketch a pre-outline of controlling ideas (topic sentences)
    \item[B] draft first, then reverse-outline to reveal the logic
    \end{itemize}
  \item Either way, check that the essay is going somewhere
  \item Expect to add, cut and rearrange: nothing is fixed until the end
  \item Use outlines to test progression and balance\\[2ex]

  \item I [FCB] normally write an outline and collect notes as I go along, then write prose at the end.  I write a rough introduction first, but revise it at the end, as I almost always change many details\ldots
\end{itemize}
\end{frame}

\begin{frame}{Introduction, Body, Conclusion}
\begin{itemize}
  \item \textbf{Introduction}: catch interest, give the context needed, state the thesis
  \item \textbf{Body}: one supporting idea per paragraph
    \begin{itemize}
    \item Transitions: each paragraph should lead to the next
    \item Support: every claim gets evidence, textual or numerical
    \end{itemize}
  \item \textbf{Conclusion}: come back to the thesis and answer: so what?
\end{itemize}
\end{frame}

\begin{frame}{Paragraphs as Structural Beams}
\begin{itemize}
  \item Each paragraph advances one controlling idea
  \item Start with a topic sentence tied to the thesis
  \item Evidence and analysis, not a list of facts
  \item End by linking forward to the next step in the argument
  \item Cut digressions
\end{itemize}
\end{frame}


\begin{frame}{Audience and Explicitness}
\begin{itemize}
  \item Define your terms; do not assume the reader shares your premises
  \item Say why the evidence matters, not just that it exists
  \item Be brief, but tell the reader where they are
  \item Clarity persuades
\end{itemize}
\end{frame}

\begin{frame}{What you should aim for}
\begin{itemize}
  \item A logical sequence that keeps moving
  \item A visible line from thesis to conclusion
  \item Every claim supported
    \begin{itemize}
    \item Reliable sources, correctly cited
    \end{itemize}
  \item The emphasis falls on the central question
\end{itemize}
\end{frame}

\begin{frame}{Writing Across Disciplines}
\begin{itemize}
  \item \textbf{Humanities:} build an argument through interpretation
    \\ → \textit{Thesis-driven essay}: claim, evidence, counterargument
  \item \textbf{Social Sciences:} explain social phenomena systematically
    \\ → \textit{IMRaD structure:} \textit{Introduction →  Methods →  Results →  Discussion}
  \item \textbf{Linguistics:} mix of humanities and science
    \\ → \textit{Intro → Background → Data/Methods → Analysis → Discussion → Conclusion}
  \item \textbf{Computer Science:} reproducibility, and what is new
    \\ → \textit{Intro → Related Work → Method → Data → Experiments/Results → Analysis → Conclusion}
  \item \textbf{Hybrids:} combine approaches (e.g., literature review + case study; policy analysis + recommendations)
\end{itemize}
\end{frame}

\begin{frame}{Types of Thesis Statements}
\begin{itemize}
  \item \textbf{Analytical}: interprets and explains evidence
        (e.g., “The novel critiques capitalism through its fragmented narration”)
  \item \textbf{Expository}: explains a concept or process
        (useful for background essays)
  \item \textbf{Argumentative}: takes a position and justifies it
        (most common in humanities writing)
  \item Choose the type that fits the essay's purpose
\end{itemize}
\end{frame}

\begin{frame}{Summary}
\begin{itemize}
  \item A strong thesis:
    \begin{itemize}
      \item Arises from inquiry
      \item Makes a specific, arguable claim
      \item Guides structure and evidence
    \end{itemize}
  \item Expect to revise it multiple times
  \item Use feedback and reflection to sharpen the argument
  \item Every paragraph should earn its place by advancing the thesis
  \item Different tasks have different goals
    \begin{itemize}
    \item You must adjust your writing style to fit the goal
    \item Different disciplines have different styles
    \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Deduction and Induction in Practice}
\begin{itemize}
  \item \textbf{Deduction}: from accepted premises to a specific conclusion
  \item[$\Rightarrow$] In real essays, premises are rarely beyond dispute, so state them
  \item \textbf{Induction}: from specific data to a generalisation; always provisional
  \item[$\Rightarrow$] Match the strength of the claim to the evidence
  \item Use both as needed; hybrid arguments are common
\end{itemize}
\end{frame}

\begin{frame}{Finding Building Blocks}
\begin{itemize}
  \item Gather: facts, quotations, data, prior interpretations
  \item Note how each item functions (example, counterexample, definition)
  \item Separate summary from analysis in notes
  \item Track source details for citation and revisiting
  \item Drop what does not advance the claim
\end{itemize}
\end{frame}

\begin{frame}{Sequencing and Emphasis}
\begin{itemize}
  \item Order points to create momentum (simple \textrightarrow{} complex)
  \item Definitions first; the fine points later
  \item Put your strongest section where it counts
  \item Use headings and transitions to signal hierarchy and shifts
  \item After drafting, try a different order
\end{itemize}
\end{frame}
